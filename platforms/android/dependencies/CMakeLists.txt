cmake_minimum_required(VERSION 3.25)
project(GemRBAndroidDependencies LANGUAGES C CXX)
if(NOT ANDROID)
  message(FATAL_ERROR "Android-only dependency build")
endif()
include(FetchContent)
file(TO_CMAKE_PATH "${DEPENDENCY_ARCHIVES}" DEPENDENCY_ARCHIVES)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
FetchContent_Declare(SDL2 URL "${DEPENDENCY_ARCHIVES}/SDL2-2.32.10.tar.gz"
  URL_HASH SHA256=5f5993c530f084535c65a6879e9b26ad441169b3e25d789d83287040a9ca5165)
FetchContent_MakeAvailable(SDL2)
install(FILES "${sdl2_SOURCE_DIR}/LICENSE.txt" DESTINATION share/licenses/SDL2)
set(ALSOFT_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ALSOFT_UTILS OFF CACHE BOOL "" FORCE)
set(ALSOFT_TESTS OFF CACHE BOOL "" FORCE)
set(ALSOFT_INSTALL_CONFIG OFF CACHE BOOL "" FORCE)
set(ALSOFT_INSTALL_HRTF_DATA OFF CACHE BOOL "" FORCE)
set(ALSOFT_INSTALL_AMBDEC_PRESETS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(openal URL "${DEPENDENCY_ARCHIVES}/openal.tar.gz"
  URL_HASH SHA256=c31b16e18179c8525e050c4798105bdbf902338701e4efc64535d4c0ff547c5c)
FetchContent_MakeAvailable(openal)
# APK native libraries must have unversioned lib*.so names and matching SONAMEs.
set_target_properties(OpenAL PROPERTIES VERSION "" SOVERSION "")
install(FILES "${openal_SOURCE_DIR}/COPYING" DESTINATION share/licenses/OpenAL)
# Build only GNU's conversion library, without Autotools/gnulib tools on Windows.
FetchContent_Declare(iconv URL "${DEPENDENCY_ARCHIVES}/iconv.tar.gz"
  URL_HASH SHA256=8f74213b56238c85a50a5329f77e06198771e70dd9a739779f4c02f65d971313)
FetchContent_MakeAvailable(iconv)
include(CheckSymbolExists)
foreach(symbol WCRTOMB MBRTOWC MBSINIT)
  string(TOLOWER "${symbol}" function)
  check_symbol_exists(${function} "wchar.h" HAVE_${symbol})
endforeach()
check_symbol_exists(CODESET "langinfo.h" HAVE_LANGINFO_CODESET)
set(DLL_VARIABLE "")
set(ICONV_CONST "")
set(EILSEQ 84)
set(USE_MBSTATE_T 1)
set(BROKEN_WCHAR_H 0)
set(HAVE_WCHAR_T 1)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/iconv-include")
configure_file("${iconv_SOURCE_DIR}/include/iconv.h.in" iconv-include/iconv.h @ONLY)
configure_file("${iconv_SOURCE_DIR}/libcharset/include/localcharset.h.in" iconv-include/localcharset.h @ONLY)
configure_file(iconv-config.h.in iconv-include/config.h)
add_library(iconv SHARED "${iconv_SOURCE_DIR}/lib/iconv.c" "${iconv_SOURCE_DIR}/libcharset/lib/localcharset.c")
target_include_directories(iconv PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/iconv-include")
install(TARGETS iconv LIBRARY DESTINATION lib)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/iconv-include/iconv.h" DESTINATION include)
install(FILES "${iconv_SOURCE_DIR}/COPYING.LIB" DESTINATION share/licenses/iconv)
